   100→/**

   102→ *
   103→ * Layout:
   104→ *   Row {
   105→ *     CoralNavRail       // 72dp vertical rail on the left

   240→    // --- Mini player position polling ---
   241→    // Polls the playback position every 500ms so the circular progress
   242→    // ring around the album art in the mini player can show song timeline.
   243→    var miniPlayerPositionMs by remember { mutableStateOf(0L) }
   244→    var miniPlayerDurationMs by remember { mutableStateOf(0L) }
   245→    // ★ Track playback duration for "Today's Top" — accurate time tracking.
   246→    //   Uses the controller's actual position to calculate how much time
   247→    //   actually elapsed since the last poll. This is ACCURATE — if you
   248→    //   played 5 minutes, it records 5 minutes, not 13.
   249→    var lastPositionMs by remember { mutableStateOf(0L) }
   250→    var lastPollTime by remember { mutableStateOf(0L) }
   251→    var trackedSongId by remember { mutableStateOf<Long?>(null) }
   252→    androidx.compose.runtime.LaunchedEffect(mediaController, isPlaying, currentSongId) {
   253→        // Reset tracking when the song changes
   254→        if (trackedSongId != currentSongId) {
   255→            trackedSongId = currentSongId
   256→            lastPositionMs = 0L
   257→            lastPollTime = System.currentTimeMillis()
   258→        }
   259→        while (true) {
   260→            try {
   261→                mediaController?.let { controller ->
   262→                    miniPlayerPositionMs = controller.currentPosition.coerceAtLeast(0L)
   263→                    miniPlayerDurationMs = controller.duration.coerceAtLeast(0L)
   264→                }
   265→                // ★ ACCURATE duration tracking — calculate actual elapsed
   266→                //   playback time using the controller's position.
   267→                //   This prevents over-counting (the 13-min bug).
   268→                if (isPlaying && currentSongId != null && !currentSongArtist.isNullOrBlank()) {
   269→                    val now = System.currentTimeMillis()
   270→                    val elapsedMs = now - lastPollTime
   271→                    lastPollTime = now
   272→                    // Only count if reasonable time elapsed (not a huge gap
   273→                    // from app being backgrounded)
   274→                    if (elapsedMs in 100..2000L) {
   275→                        val elapsedSeconds = (elapsedMs / 1000L).toInt()
   276→                        if (elapsedSeconds > 0) {
   277→                            com.rajatxo.coral.data.prefs.PlaybackHistory.addPlayDuration(
   278→                                songId = currentSongId,
   279→                                artist = currentSongArtist,
   280→                                secondsToAdd = elapsedSeconds
   281→                            )
   282→                        }
   283→                    }
   284→                }
   285→            } catch (_: Exception) { }
   286→            kotlinx.coroutines.delay(if (isPlaying) 1000L else 2000L)
   287→        }
   288→    }
   289→
   290→    // Equalizer controller + sleep timer — singletons for the home screen's

   292→    // survives config changes and isn't reset when the screen recomposes.
   293→    val equalizerController = remember { com.rajatxo.coral.audio.EqualizerController() }
   294→    // SleepTimer is now a SINGLETON — no need to create it here.
   295→    // It survives Activity destruction (rotation, task manager kill).
   296→    val sleepTimer = com.rajatxo.coral.data.premium.SleepTimer
   297→
   298→    // When the current song changes, fire the sleep timer's end-of-song
   299→    // trigger (in case the user set the timer to "end of current song").

   301→        if (currentSongId != null) {
   302→            sleepTimer.onSongEnd()
   303→        }
   304→    }
   305→
   306→    // --- Sleep Timer Capsule state ---
   307→    val sleepTimerState by sleepTimer.state.collectAsState()
   308→    var sleepRemainingMs by remember { mutableStateOf(0L) }
   309→
   310→    // When the timer fires (active goes true→false), pause playback
   311→    androidx.compose.runtime.LaunchedEffect(sleepTimerState.active) {
   312→        if (!sleepTimerState.active && sleepRemainingMs > 0) {
   313→            // Timer just expired — pause the player
   314→            onSongEnded()
   315→            sleepRemainingMs = 0
   316→        }
   317→    }
   318→
   319→    // Poll remaining time every 1 second when timer is active
   320→    androidx.compose.runtime.LaunchedEffect(sleepTimerState.active) {
   321→        while (sleepTimerState.active) {
   322→            val endAt = sleepTimerState.endAtMs
   323→            if (endAt != null) {
   324→                sleepRemainingMs = (endAt - System.currentTimeMillis()).coerceAtLeast(0L)
   325→            }
   326→            delay(1000L)
   327→        }
   328→    }
   329→

   335→    val homeContext = androidx.compose.ui.platform.LocalContext.current

   354→        }

   396→        // dismissed (fades out + slides away) and playback pauses.
   397→        //
   398→        // The mini player reappears when:
   399→        //   • A new song is selected (currentSongId changes) — LaunchedEffect

   402→        //     Without this, replaying the same song (currentSongId stays
   403→        //     the same) wouldn't trigger the LaunchedEffect, leaving the

   411→                miniPlayerDismissed = false
   412→            }
   413→        }

   415→        //   Used by the Speed Dial "Based on: Last Played" mode to bias
   416→        //   the grid toward songs by the most-recently-played artist.

   418→            if (currentSongId != null && !currentSongArtist.isNullOrBlank()) {
   419→                com.rajatxo.coral.data.prefs.PlaybackHistory.recordPlayback(
   420→                    songId = currentSongId,
   421→                    artist = currentSongArtist
   422→                )
   423→                // ★ Record daily play count for "Today's Top"
   424→                com.rajatxo.coral.data.prefs.PlaybackHistory.recordDailyPlay(
   425→                    songId = currentSongId,
   426→                    artist = currentSongArtist
   427→                )
   428→            }
   429→        }
   430→        val onSongClickWithReset: (Song) -> Unit = { song ->
   431→            miniPlayerDismissed = false
   432→            onSongClick(song)
   433→        }
   434→        val onSongClickWithQueueReset: (Song, List<Song>) -> Unit = { song, queue ->
   435→            miniPlayerDismissed = false
   436→            onSongClickWithQueue(song, queue)
   437→        }

   439→        // ─── Push back animation (Yuma-style) ──
   440→        // When the FullPlayer opens, the main content dims (alpha → 0.6)
   441→        // — like the home screen is being "pushed back" behind the player.
   442→        //
   443→        // NOTE: Previously also scaled down (0.93x) but that exposed black
   444→        // bars on both sides of the screen during the animation (the
   445→        // scaled-down content didn't cover the full width, showing the
   446→        // parent's background). Removed the scale — alpha-only dimming
   447→        // still gives the "push back" feel without the visual glitch.
   448→        val pushBackFraction by animateFloatAsState(
   449→            targetValue = if (showFullPlayer) 1f else 0f,
   450→            animationSpec = tween(350),
   451→            label = "pushBack"
   452→        )
   453→        Box(
   454→            modifier = Modifier
   455→                .fillMaxSize()
   456→                .graphicsLayer {
   457→                    alpha = 1f - (0.4f * pushBackFraction)
   458→                }
   459→                .layerBackdrop(glassBackdrop)
   460→        ) {
   461→            when (selectedTab) {
   462→                CoralTab.QuickPicks -> QuickPicksScreen(
   463→                    songs = songs,
   464→                    currentSongId = currentSongId,
   465→                    currentSongArt = currentSongArt,
   466→                    capsuleVisible = capsuleVisible,
   467→                    capsuleRemaining = capsuleRemaining,
   468→                    onExtend = onExtend,
   469→                    onSongClick = onSongClickWithReset,
   470→                    isPlaying = isPlaying,
   471→                    onPlayPauseClick = onPlayPauseClick,

  1297→@Composable
  1298→private fun MiniPlayer(
  1299→    title: String,
  1300→    artist: String,
  1301→    albumArtUri: android.net.Uri?,
  1302→    songId: Long?,
  1303→    isPlaying: Boolean,
  1304→    positionMs: Long,
  1305→    durationMs: Long,
  1306→    onPlayPauseClick: () -> Unit,
  1307→    onNextClick: () -> Unit,
  1308→    onClick: () -> Unit,
  1309→    onSwipeUp: () -> Unit = {},
  1310→    onSwipeDismiss: () -> Unit = {},
  1311→    isFullPlayerOpen: Boolean = false,
  1312→    backdrop: LayerBackdrop? = null
  1313→) {
  1314→    // ★ Custom cover override — if the song has a custom cover set in
  1315→    //   SongCoverManager, show THAT in the mini player instead of the
  1316→    //   file's original album art.
  1317→    //
  1318→    //   ★★ CRITICAL: rememberEffectiveCover() observes SongCoverManager.revision
  1319→    //   so the mini player recomposes immediately when the user sets/resets
  1320→    //   a custom cover for the CURRENTLY PLAYING song. Without this, the
  1321→    //   mini player wouldn't update (songId stays the same, so plain
  1322→    //   remember(songId) wouldn't re-execute).
  1323→    val effectiveAlbumArtUri = com.rajatxo.coral.util.rememberEffectiveCover(
  1324→        songId, albumArtUri
  1325→    )
  1326→
  1327→    // Notched mini player — pill with a U-shaped concave notch at the
  1328→    // bottom-center, and the search capsule nested inside that notch.
  1329→    //
  1330→    // Layout (top to bottom):
  1331→    //   [Album art + progress ring] [Title + artist] [Heart]
  1332→    //   [U-notch with search capsule nested inside]
  1333→    //
  1334→    // The NotchedPillShape clips the entire body so the bottom edge has
  1335→    // the U-curve cutout. The search capsule is positioned at the bottom,
  1336→    // overlapping the notch — top half recessed, bottom half hanging below.
  1337→    //
  1338→    // BACKGROUND: Real frosted-glass blur via the same kyant/backdrop
  1339→    // library the nav bar uses (drawBackdrop + AGSL blur). The mini player
  1340→    // now samples whatever is behind it on the home screen — song list,
  1341→    // album art, etc. — and blurs it in real time. Replaces the old
  1342→    // 'blurred album cover + dark tint' background. Shape is unchanged
  1343→    // (still a 32dp rounded pill, same border).
  1344→    //
  1345→    // GESTURES (Yuma-inspired, original implementation — not copied):
  1346→    //   • Tap → opens FullPlayer (existing behavior, kept)
  1347→    //   • Swipe UP → opens FullPlayer (same as tap, but gesture-based)
  1348→    //   • Swipe LEFT/RIGHT → dismisses the mini player (slides out + fades)
  1349→    //     and pauses playback. The mini player reappears when a new song
  1350→    //     is selected.
  1351→    //
  1352→    // The gesture uses a single detectDragGestures pointerInput. On drag
  1353→    // start, no direction is locked. Once the drag exceeds 20px in either
  1354→    // axis, the direction locks (horizontal or vertical). This prevents
  1355→    // diagonal drags from triggering both actions.
  1356→
  1357→    val favorites by com.rajatxo.coral.data.store.PlaylistStore.favorites.collectAsState()
  1358→    val isFavorite = songId != null && songId in favorites.songIds
  1359→
  1360→    val pillShape: Shape = RoundedCornerShape(32.dp)
  1361→
  1362→    // ─── Gesture state ──────────────────────────────────────────────
  1363→    val scope = rememberCoroutineScope()
  1364→    val density = androidx.compose.ui.platform.LocalDensity.current
  1365→    val screenWidthPx = with(density) {
  1366→        androidx.compose.ui.platform.LocalConfiguration.current.screenWidthDp.dp.toPx()
  1367→    }
  1368→
  1369→    val offsetX = remember { Animatable(0f) }
  1370→    val offsetY = remember { Animatable(0f) }
  1371→    val scale = remember { Animatable(1f) }
  1372→
  1373→    // Max distance the mini player can be dragged upward (in pixels).
  1374→    // 60dp — keeps the gesture short (no dragging all the way to the top).
  1375→    // The mini player fades out as it approaches this cap, so by 60dp up
  1376→    // it's fully dissolved — the full player then opens.
  1377→    val maxSwipeUpPx = with(density) { 60.dp.toPx() }
  1378→
  1379→    // ─── Reset offsets when the full player closes ──────────────────
  1380→    // When the user swipes up to open the full player, the mini player
  1381→    // fades away (offsetY animates up + alpha fades). We DON'T reset
  1382→    // offsetY to 0 immediately — that would cause a "splash" flash
  1383→    // (mini player snapping back to position before the full player
  1384→    // covers it).
  1385→    //
  1386→    // Instead, we leave offsetY at the faded position while the full
  1387→    // player is open (the mini player is invisible anyway — it's under
  1388→    // the full player). When the full player CLOSES (isFullPlayerOpen
  1389→    // goes true → false), this LaunchedEffect fires and resets offsetY
  1390→    // to 0 so the mini player reappears at its normal position.
  1391→    androidx.compose.runtime.LaunchedEffect(isFullPlayerOpen) {
  1392→        if (!isFullPlayerOpen) {
  1393→            scope.launch {
  1394→                offsetY.snapTo(0f)
  1395→                offsetX.snapTo(0f)
  1396→                scale.snapTo(1f)
  1397→            }
  1398→        }
  1399→    }
  1400→
  1401→    // Drag direction lock — once the drag exceeds the threshold, we
  1402→    // commit to either HORIZONTAL (dismiss) or VERTICAL (expand).
  1403→    var dragDirection: Int? by remember { mutableStateOf(null) }  // 0=H, 1=V
  1404→    var totalDragX by remember { mutableStateOf(0f) }
  1405→    var totalDragY by remember { mutableStateOf(0f) }
  1406→
  1407→    Box(
  1408→        modifier = Modifier
  1409→            .padding(vertical = 4.dp)
  1410→            .consistentNavBarPadding()
  1411→            .graphicsLayer {
  1412→                translationX = offsetX.value
  1413→                translationY = offsetY.value.coerceAtLeast(-maxSwipeUpPx)
  1414→                val s = scale.value
  1415→                scaleX = s
  1416→                scaleY = s
  1417→                alpha = (1f - abs(offsetX.value) / screenWidthPx).coerceIn(0f, 1f) *
  1418→                    (1f - (abs(offsetY.value) / maxSwipeUpPx)).coerceIn(0f, 1f)
  1419→            }
  1420→            .pointerInput(Unit) {
  1421→                detectDragGestures(
  1422→                    onDragStart = {
  1423→                        dragDirection = null
  1424→                        totalDragX = 0f
  1425→                        totalDragY = 0f
  1426→                        scope.launch { scale.snapTo(0.96f) }
  1427→                    },
  1428→                    onDrag = { change, dragAmount ->
  1429→                        change.consume()
  1430→                        totalDragX += dragAmount.x
  1431→                        totalDragY += dragAmount.y
  1432→
  1433→                        // Lock direction once the drag exceeds 20px
  1434→                        if (dragDirection == null) {
  1435→                            if (abs(totalDragX) > 20f || abs(totalDragY) > 20f) {
  1436→                                dragDirection = if (abs(totalDragX) > abs(totalDragY)) 0 else 1
  1437→                            }
  1438→                        }
  1439→
  1440→                        when (dragDirection) {
  1441→                            0 -> {  // HORIZONTAL — move the mini player with the finger
  1442→                                scope.launch { offsetX.snapTo(totalDragX) }
  1443→                            }
  1444→                            1 -> {  // VERTICAL — only follow upward drags (swipe up)
  1445→                                if (totalDragY < 0) {
  1446→                                    // Dampened follow — mini player moves at 0.4x the drag speed
  1447→                                    // (was 0.5x — slower, so the 60dp cap is reached with a
  1448→                                    // more deliberate gesture). Combined with the alpha fade,
  1449→                                    // this gives a smooth "lift up and dissolve" feel.
  1450→                                    scope.launch { offsetY.snapTo(totalDragY * 0.4f) }
  1451→                                }
  1452→                            }
  1453→                        }
  1454→                    },
  1455→                    onDragEnd = {
  1456→                        scope.launch { scale.animateTo(1f, spring(dampingRatio = Spring.DampingRatioMediumBouncy)) }
  1457→                        when (dragDirection) {
  1458→                            0 -> {  // HORIZONTAL
  1459→                                if (abs(totalDragX) > screenWidthPx * 0.4f) {
  1460→                                    // Past the 40% threshold → dismiss
  1461→                                    val target = if (totalDragX < 0) -screenWidthPx else screenWidthPx
  1462→                                    scope.launch {
  1463→                                        offsetX.animateTo(target, tween(200))
  1464→                                        onSwipeDismiss()
  1465→                                        delay(100)
  1466→                                        offsetX.snapTo(0f)
  1467→                                    }
  1468→                                } else {
  1469→                                    // Not past threshold → spring back
  1470→                                    scope.launch {
  1471→                                        offsetX.animateTo(0f, spring(dampingRatio = Spring.DampingRatioMediumBouncy))
  1472→                                    }
  1473→                                }
  1474→                            }
  1475→                            1 -> {  // VERTICAL
  1476→                                // If swiped up at least 30px → open FullPlayer.
  1477→                                // Lower threshold than before (was 100px) — since the
  1478→                                // visual is now a short lift + fade (not a long drag),
  1479→                                // a smaller swipe should trigger it.
  1480→                                if (totalDragY < -30f) {
  1481→                                    // Animate the mini player fading away smoothly,
  1482→                                    // then open the full player. The 200ms tween gives
  1483→                                    // a buttery dissolve before the player opens.
  1484→                                    //
  1485→                                    // NOTE: Do NOT reset offsetY to 0 here. The mini
  1486→                                    // player stays translated up (invisible) while
  1487→                                    // the full player is open. When the full player
  1488→                                    // closes, the LaunchedEffect(isFullPlayerOpen)
  1489→                                    // above resets offsetY to 0 so the mini player
  1490→                                    // reappears at its normal position.
  1491→                                    //
  1492→                                    // Previously, offsetY.snapTo(0f) here caused a
  1493→                                    // "splash" flash — the mini player snapped back
  1494→                                    // to position before the full player had fully
  1495→                                    // covered the screen.
  1496→                                    scope.launch {
  1497→                                        offsetY.animateTo(-maxSwipeUpPx * 2f, tween(200))
  1498→                                        onSwipeUp()
  1499→                                        // Don't reset offsetY — the LaunchedEffect
  1500→                                        // handles it when the full player closes.
  1501→                                    }
  1502→                                } else {
  1503→                                    // Not enough swipe → spring back
  1504→                                    scope.launch {
  1505→                                        offsetY.animateTo(0f, spring(dampingRatio = Spring.DampingRatioMediumBouncy))
  1506→                                    }
  1507→                                }
  1508→                            }
  1509→                        }
  1510→                    },
  1511→                    onDragCancel = {
  1512→                        scope.launch {
