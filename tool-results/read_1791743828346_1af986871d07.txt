    12→/**
    13→ * Coral's custom icon set.
    14→ *
    15→ * All glyphs are hand-drawn vector paths in the Lucide (MIT) visual idiom:
    16→ *  - 24 x 24 viewport
    17→ *  - 2dp stroke width
    18→ *  - round caps & joins
    19→ *  - filled variants use SolidColor fill
    20→ *
    21→ * Keeping these in-house (instead of pulling Material Icons or Lucide as a
    22→ * dependency) keeps Coral's dependency tree 100 % permissively licensed and
    23→ * avoids the GPL-tainted Material Icons asset pack.
    24→ */
    25→object CoralIcons {
    26→
    27→    private fun stroke(
    28→        name: String,
    29→        pathBuilder: PathBuilder.() -> Unit
    30→    ): ImageVector = ImageVector.Builder(
    31→        name = name,
    32→        defaultWidth = 24.dp,
    33→        defaultHeight = 24.dp,
    34→        viewportWidth = 24f,
    35→        viewportHeight = 24f
    36→    ).apply {
    37→        path(
    38→            fill = null,
    39→            stroke = SolidColor(Color.Black),
    40→            strokeLineWidth = 2f,
    41→            strokeLineCap = StrokeCap.Round,
    42→            strokeLineJoin = StrokeJoin.Round,
    43→            strokeLineMiter = 4f,
    44→            pathBuilder = pathBuilder
    45→        )
    46→    }.build()
