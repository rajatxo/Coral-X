   110→/**
   111→ * QuickPicksScreen — "Editorial Gallery" edition.
   112→ *
   113→ * Light/dark theme-aware. Asymmetric hero grid + horizontal carousels.
   114→ * Songs are randomized on each app launch. Carousels are infinite
   115→ * (repeat the song list so you can swipe forever).
   116→ *
   117→ * Pull-to-refresh: pulling down on this page refreshes the Quick Picks
   118→ * content (rolls a new random seed → new hero/recent/more selections,
   119→ * same effect as tab-switching away and back). The indicator is a soft
   120→ * wind animation, NOT the boring circular arrow.
   121→ */
   122→@OptIn(ExperimentalMaterial3Api::class)
   123→@Composable
   124→fun QuickPicksScreen(
   125→    songs: List<Song>,
   126→    currentSongId: Long?,
   127→    currentSongArt: android.net.Uri? = null,
   128→    capsuleVisible: Boolean = false,
   129→    capsuleRemaining: Long = 0L,
   130→    onExtend: () -> Unit = {},
   131→    onSongClick: (Song) -> Unit = {},
   132→    onBackClick: () -> Unit = {},
   133→    isPlaying: Boolean = false,
   134→    onPlayPauseClick: () -> Unit = {},
   135→    onCapsuleClick: (Song) -> Unit = {},
   136→    onSpeedDialHeaderClick: () -> Unit = {},
   137→    glassHazeState: dev.chrisbanes.haze.HazeState? = null,
   138→    glassStyle: dev.chrisbanes.haze.HazeStyle? = null
   139→) {
   140→    val context = LocalContext.current
   141→
   142→    // ─── Current song's palette → dark gradient background ──────────
   143→    // The background is a dark gradient using the current song's palette
   144→    // colors. If no song is playing, use a default dark palette.
   145→    var palette by remember { mutableStateOf(CoralPalette.Default) }
   146→    LaunchedEffect(currentSongArt) {
   147→        if (currentSongArt != null) {
   148→            PaletteCache.get(currentSongArt)?.let { palette = it }
   149→            extractPalette(context, currentSongArt)?.let {
   150→                palette = it
   151→                PaletteCache.put(currentSongArt, it)
   152→            }
   153→        }
   154→    }
   155→    // ─── Background gradient (single dominant color → dark) ─────────
   156→    // Uses ONLY the dominant palette color (palette.primary), not a mix
   157→    // of primary + secondary. One color, fading from vibrant at the top
   158→    // (behind the blur header) to near-black at the bottom.
   159→    //
   160→    // The transition from vibrant → dark is BUTTERY SMOOTH via many
   161→    // closely-spaced color stops. Each stop smoothly steps the alpha
   162→    // down, so there's no visible "band" or hard transition line.
   163→    //
   164→    // Layout:
   165→    //   0.00 - 0.10  → vibrant (alpha 0.85)  [behind blur header]
   166→    //   0.10 - 0.30  → buttery smooth fade   [transition zone]
   167→    //   0.30 - 1.00  → dark gradient         [page content]
   168→    //
   169→    // All colors animate smoothly when the song changes (tween 800ms).
   170→    val vibrantTop by animateColorAsState(
   171→        targetValue = palette.primary.copy(alpha = 0.85f),
   172→        animationSpec = tween(800), label = "bgVibrantTop"
   173→    )
   174→    // Mid-transition stops — same hue, progressively lower alpha.
   175→    // These create the "buttery" feel by stepping alpha down gradually
   176→    // instead of one hard jump from 0.85 → 0.25.
   177→    val fade1 by animateColorAsState(
   178→        targetValue = palette.primary.copy(alpha = 0.65f),
   179→        animationSpec = tween(800), label = "bgFade1"
   180→    )
   181→    val fade2 by animateColorAsState(
   182→        targetValue = palette.primary.copy(alpha = 0.45f),
   183→        animationSpec = tween(800), label = "bgFade2"
   184→    )
   185→    val fade3 by animateColorAsState(
   186→        targetValue = palette.primary.copy(alpha = 0.28f),
   187→        animationSpec = tween(800), label = "bgFade3"
   188→    )
   189→    val animatedTop by animateColorAsState(
   190→        targetValue = palette.primary.copy(alpha = 0.18f),
   191→        animationSpec = tween(800), label = "bgTop"
   192→    )
   193→    val animatedMid by animateColorAsState(
   194→        targetValue = palette.primary.copy(alpha = 0.08f),
   195→        animationSpec = tween(800), label = "bgMid"
   196→    )
   197→    val animatedBottom by animateColorAsState(
   198→        targetValue = Color(0xFF05050A),  // near-black at the bottom
   199→        animationSpec = tween(800), label = "bgBottom"
   200→    )
   201→
   202→    // Dark text colors (always white on the dark gradient)
   203→    val textPrimary = Color.White
   204→    val textSecondary = Color.White.copy(alpha = 0.6f)
   205→
   206→    // ─── Random seed — changes on every app launch OR pull-to-refresh ─
   207→    // This ensures the song selection is different each time the user
   208→    // opens the app OR pulls to refresh the Quick Picks page.
   209→    // Keyed on refreshKey so bumping it (via PTR) rolls a new seed,
   210→    // which cascades into heroSongs / recentSongs / moreSongs below.
   211→    var refreshKey by remember { mutableIntStateOf(0) }
   212→    val launchSeed = remember(refreshKey) { Random.nextInt() }
   213→
   214→    // ─── Pull-to-refresh state ───────────────────────────────────────
   215→    // Pulling down on this page refreshes the Quick Picks content
   216→    // (rolls a new launchSeed → new random song selections, same
   217→    // effect as tab-switching away and back).
   218→    //
   219→    // The default Material3 circular arrow indicator is suppressed
   220→    // (indicator = {} on PullToRefreshBox). Instead, a custom
   221→    // BugLineRefreshIndicator is rendered in the Speed Dial header
   222→    // — right next to the chevron. A bug icon crawls along a thin line
   223→    // as the user pulls, rotating as it moves.
   224→    val ptrState: PullToRefreshState = rememberPullToRefreshState()
   225→    var isRefreshing by remember { mutableStateOf(false) }
   226→    val scope = rememberCoroutineScope()
   227→
   228→    // ─── Prepare song groups (randomized per launch) ────────────────
   229→    // BUG FIX: previously used remember(songs, currentSongId) which
   230→    // didn't recompute when songs loaded async after the screen was
   231→    // already shown. Now we key on songs.size so it recomputes when
   232→    // songs actually arrive. Also use launchSeed for randomization so
   233→    // the selection changes on every app open.
   234→    val heroSongs = remember(songs.size, currentSongId, launchSeed) {
   235→        if (songs.isEmpty()) emptyList()
   236→        else {
   237→            val current = songs.firstOrNull { it.id == currentSongId }
   238→            val pool = songs.shuffled(Random(launchSeed))
   239→            if (current != null) {
   240→                listOf(current, pool.firstOrNull { it.id != current.id } ?: songs.first())
   241→            } else {
   242→                pool.take(2)
   243→            }
   244→        }
   245→    }
   246→    val recentSongs = remember(songs.size) {
   247→        if (songs.isEmpty()) emptyList()
   248→        else {
   249→            // 50 most-recently-ADDED songs (newest first). dateAdded is
   250→            // Unix epoch seconds from MediaStore.Audio.Media.DATE_ADDED.
   251→            // Falls back to original insertion order if all dateAdded are 0
   252→            // (cache saved before this field existed -- sort is then a no-op).
   253→            songs.sortedByDescending { it.dateAdded }.take(50)
   254→        }
   255→    }
   256→    val moreSongs = remember(songs.size, launchSeed) {
   257→        if (songs.isEmpty()) emptyList()
   258→        else songs.shuffled(Random(launchSeed + 2)).take(50)
   259→    }
   260→
   261→    // Recent and More rows now have 50 unique songs each -- no need to
   262→    // infinitely repeat the same 10 songs 3x. The LazyRow scrolls through
   263→    // 50 distinct items directly. Kept the variable names for compat with
   264→    // the LazyRow items() calls below.
   265→    val infiniteRecent = recentSongs
   266→    val infiniteMore = moreSongs
   267→
   268→    // Solid dark base — the gradient's low-alpha palette colors composite
   269→    // over this, so the background is always predominantly dark/black
   270→    // EXCEPT in the top zone where the vibrant colors are opaque enough
   271→    // to fully cover the darkBase.
   272→    val darkBase = Color(0xFF05050A)
   273→
   274→    PullToRefreshBox(
   275→        isRefreshing = isRefreshing,
   276→        onRefresh = {
   277→            // Refresh the Quick Picks page: bump refreshKey, which rolls
   278→            // a new launchSeed → new heroSongs/recentSongs/moreSongs
   279→            // selections.
   280→            isRefreshing = true
   281→            refreshKey++
   282→            scope.launch {
   283→                delay(900)
   284→                isRefreshing = false
   285→            }
   286→        },
   287→        state = ptrState,
   288→        modifier = Modifier
   289→            .fillMaxSize()
   290→            .background(darkBase)
   291→            .background(
   292→                // Single dominant color → dark gradient, buttery smooth.
   293→                // 8 stops total — closely spaced for a continuous fade
   294→                // with no visible banding or hard transitions.
   295→                //   0.00  vibrant (0.85)   — behind blur header
   296→                //   0.10  fade1  (0.65)   ┐
   297→                //   0.15  fade2  (0.45)   │ transition zone (buttery)
   298→                //   0.20  fade3  (0.28)   ┘
   299→                //   0.30  top    (0.18)   — dark gradient kicks in
   300→                //   0.55  mid    (0.08)
   301→                //   1.00  bottom (black)
   302→                Brush.verticalGradient(
   303→                    colorStops = arrayOf(
   304→                        0.0f  to vibrantTop,
   305→                        0.10f to fade1,
   306→                        0.15f to fade2,
   307→                        0.20f to fade3,
   308→                        0.30f to animatedTop,
   309→                        0.55f to animatedMid,
   310→                        1.0f  to animatedBottom
   311→                    )
   312→                )
   313→            ),
   314→        // Suppress the default Material3 circular arrow indicator — we
   315→        // use our custom BugLineRefreshIndicator in the Speed Dial header
   316→        // instead (positioned right next to the chevron).
   317→        indicator = {}
   318→    ) {
   319→        LazyColumn(
   320→            modifier = Modifier
   321→                .fillMaxSize(),
   322→            contentPadding = PaddingValues(
   323→                top = 108.dp,      // just below where the blur ends
   324→                bottom = 200.dp,
   325→                start = 20.dp,
   326→                end = 20.dp
   327→            ),
   328→            verticalArrangement = Arrangement.spacedBy(6.dp)
   329→        ) {
   330→            // ═══ Greeting + Mood ═══
   331→            item {
   332→                GreetingMoodHeader(
   333→                    textPrimary = textPrimary,
   334→                    textSecondary = textSecondary,
   335→                    modifier = Modifier.padding(bottom = 6.dp),
   336→                    isPlaying = isPlaying
   337→                )
   338→            }
   339→
   340→            // ═══ Today's Top — Instagram-story style capsules ═══
   341→            // 3 capsule stories with album covers + gradient borders.
   342→            // Tap a COVER inside a capsule → opens the floating
   343→            // miniplayer-style TodaysTopCard for that specific song.
   344→            item {
   345→                TodaysTopSection(
   346→                    songs = songs,
   347→                    textPrimary = textPrimary,
   348→                    textSecondary = textSecondary,
   349→                    onCapsuleClick = onCapsuleClick
   350→                )
   351→            }
   352→
   353→            // ═══ Speed Dial (first row) ═══
   354→            // A paginated grid of square song cards + a "randomize" dice
   355→            // button as the last slot. Tap a card to play that song.
   356→            // Tap the dice → plays a random song.
   357→            // The hero grid (EditorialCard row) has been removed — the
   358→            // Speed Dial is now the first row on Quick Picks.
   359→            item {
   360→                SpeedDialSection(
   361→                    songs = songs,
   362→                    currentSongId = currentSongId,
   363→                    onSongClick = onSongClick,
   364→                    textPrimary = textPrimary,
   365→                    textSecondary = textSecondary,
   366→                    launchSeed = launchSeed,
   367→                    pullProgress = ptrState.distanceFraction,
   368→                    isRefreshing = isRefreshing,
   369→                    onSpeedDialHeaderClick = onSpeedDialHeaderClick,
   370→                    glassHazeState = glassHazeState,
   371→                    glassStyle = glassStyle
   372→                )
   373→            }
   374→
   375→            // ═══ Recent ═══
   376→            item {
   377→                SectionHeader(
   378→                    title = "Recent",
   379→                    count = recentSongs.size,
   380→                    textPrimary = textPrimary,
   381→                    textSecondary = textSecondary
   382→                )
   383→            }
   384→            item {
   385→                LazyRow(
   386→                    horizontalArrangement = Arrangement.spacedBy(12.dp),
   387→                    contentPadding = PaddingValues(horizontal = 0.dp)
   388→                ) {
   389→                    itemsIndexed(
   390→                        items = infiniteRecent,
   391→                        // Stable key = song id + occurrence index, so each
   392→                        // repeated copy of a song has a unique key. Without
   393→                        // a key, LazyRow recomposes ALL items on every
   394→                        // songs update instead of reusing existing ones.
   395→                        key = { index, song -> "${song.id}_$index" }
   396→                    ) { _, song ->
   397→                        SquareCard(
   398→                            song = song,
   399→                            isCurrent = song.id == currentSongId,
   400→                            onClick = { onSongClick(song) },
   401→                            modifier = Modifier
   402→                                .width(140.dp)
   403→                                .height(200.dp)
   404→                        )
   405→                    }
   406→                }
   407→            }
   408→
   409→            // ═══ More ═══
   410→            item {
   411→                SectionHeader(
   412→                    title = "More picks",
   413→                    count = moreSongs.size,
   414→                    textPrimary = textPrimary,
   415→                    textSecondary = textSecondary
   416→                )
   417→            }
   418→            item {
   419→                LazyRow(
   420→                    horizontalArrangement = Arrangement.spacedBy(12.dp),
   421→                    contentPadding = PaddingValues(horizontal = 0.dp)
   422→                ) {
   423→                    itemsIndexed(
   424→                        items = infiniteMore,
   425→                        key = { index, song -> "${song.id}_$index" }
   426→                    ) { _, song ->
   427→                        LandscapeCard(
   428→                            song = song,
   429→                            isCurrent = song.id == currentSongId,
   430→                            onClick = { onSongClick(song) },
   431→                            modifier = Modifier
   432→                                .width(160.dp)
   433→                                .height(220.dp)
   434→                        )
   435→                    }
   436→                }
   437→            }
   438→        }
   439→
   440→    }
   441→}

   446→
   447→@Composable
   448→private fun EditorialCard(
   449→    song: Song,
   450→    isCurrent: Boolean,
   451→    onClick: () -> Unit,
   452→    modifier: Modifier = Modifier
   453→) {
   454→    val cardShape = RoundedCornerShape(24.dp)
   455→    Box(
   456→        modifier = modifier
   457→            .shadow(
   458→                elevation = 8.dp,
   459→                shape = cardShape,
   460→                clip = false
   461→            )
   462→            .clip(cardShape)
   463→            .clickable(
   464→                interactionSource = remember { MutableInteractionSource() },
   465→                indication = null,
   466→                onClick = onClick
   467→            )
   468→    ) {
   469→        // ═══ Spiral 2.0-style album art blur-blend ═══
   470→        //
   471→        // Layer 1 (bottom): BLURRED album art — fills the entire card.
   472→        // This is the "down" part. Heavy blur so it's just soft colors.
   473→        if (com.rajatxo.coral.util.SongCoverManager.getEffectiveCover(song.id, song.albumArtUri) != null) {
   474→            AsyncImage(
   475→                model = com.rajatxo.coral.util.SongCoverManager.getEffectiveCover(song.id, song.albumArtUri),
   476→                contentDescription = null,
   477→                contentScale = ContentScale.Crop,
   478→                modifier = Modifier
   479→                    .fillMaxSize()
   480→                    .blur(48.dp)
   481→            )
   482→        } else {
   483→            Box(
   484→                modifier = Modifier
   485→                    .fillMaxSize()
   486→                    .background(Color(0xFF3A3A3C)),
   487→                contentAlignment = Alignment.Center
   488→            ) {
   489→                Icon(
   490→                    imageVector = CoralIcons.Music,
   491→                    contentDescription = null,
   492→                    tint = Color.White.copy(alpha = 0.5f),
   493→                    modifier = Modifier.size(32.dp)
   494→                )
   495→            }
   496→        }
   497→
   498→        // Layer 2 (top): SHARP album art — covers the top ~75% of the card.
   499→        // A DstIn gradient at its bottom edge dissolves the sharp image
   500→        // into the blurred layer behind it (the "blend point").
   501→        // Smaller blend height = less gap, more of the sharp cover shows.
   502→        if (com.rajatxo.coral.util.SongCoverManager.getEffectiveCover(song.id, song.albumArtUri) != null) {
   503→            Box(
   504→                modifier = Modifier
   505→                    .fillMaxWidth()
   506→                    .fillMaxHeight(0.75f)
   507→                    .align(Alignment.TopCenter)
   508→                    .graphicsLayer {
   509→                        compositingStrategy = CompositingStrategy.Offscreen
   510→                    }
   511→                    .drawWithContent {
   512→                        drawContent()
   513→                        // DstIn mask: opaque at top → transparent at bottom.
   514→                        // Smaller blend height (40dp) for a tighter blend.
   515→                        val blendHeightPx = 40.dp.toPx()
   516→                        val imageHeight = size.height
   517→                        val blendStartY = (imageHeight - blendHeightPx).coerceAtLeast(0f)
   518→                        drawRect(
   519→                            brush = Brush.verticalGradient(
   520→                                colors = listOf(Color.Black, Color.Transparent),
   521→                                startY = blendStartY,
   522→                                endY = imageHeight
   523→                            ),
   524→                            blendMode = BlendMode.DstIn
   525→                        )
   526→                    }
   527→            ) {
   528→                AsyncImage(
   529→                    model = com.rajatxo.coral.util.SongCoverManager.getEffectiveCover(song.id, song.albumArtUri),
   530→                    contentDescription = null,
   531→                    contentScale = ContentScale.Crop,

   840→// ════════════════════════════════════════════════════════════════════
   841→// SPEED DIAL SECTION
   842→// ════════════════════════════════════════════════════════════════════
   843→// A paginated grid of square song cards + a "randomize" dice button.
   844→// Based on vivi-music's SpeedDial feature:
   845→//   - HorizontalPager with pages of a 3-column grid
   846→//   - Each page shows up to 9 songs (3x3)
   847→//   - The last slot on the first page is a "randomize" dice button
   848→//   - Tap a song card → plays that song
   849→//   - Tap the dice → picks a random song and plays it
   850→// ════════════════════════════════════════════════════════════════════
   851→
   852→@Composable
   853→private fun SpeedDialSection(
   854→    songs: List<Song>,
   855→    currentSongId: Long?,
   856→    onSongClick: (Song) -> Unit,
   857→    textPrimary: Color,
   858→    textSecondary: Color,
   859→    launchSeed: Int,
   860→    pullProgress: Float = 0f,
   861→    isRefreshing: Boolean = false,
   862→    onSpeedDialHeaderClick: () -> Unit = {},
   863→    glassHazeState: dev.chrisbanes.haze.HazeState? = null,
   864→    glassStyle: dev.chrisbanes.haze.HazeStyle? = null
   865→) {
   866→    val scope = rememberCoroutineScope()
   867→    var isRandomizing by remember { mutableStateOf(false) }
   868→
   869→    // Observe pinned song IDs so cards re-render when a song is pinned/unpinned.
   870→    val pinnedIds by com.rajatxo.coral.data.prefs.SpeedDialPinStore.pinnedIds.collectAsState()
   871→
   872→    // ─── Build the Speed Dial song list ──
   873→    //
   874→    // Two modes (user-selectable via the "Based on" capsule popup):
   875→    //   - RANDOM      — completely random songs (original behavior)
   876→    //   - LAST_PLAYED — 80% songs from the last-played artist + 20% other.
   877→    //                   Falls back to RANDOM if no playback history exists.
   878→    //
   879→    // Pinned songs come FIRST (in the order they were pinned — Set preserves
   880→    // insertion order for LinkedHashSet, which is what SharedPreferences
   881→    // StringSet returns after our save). They sit at the top-left of the
   882→    // grid and stay there across refresh/restart.
   883→    val speedDialMode by com.rajatxo.coral.data.prefs.SpeedDialModeManager.mode.collectAsState()
   884→    val playbackHistory by com.rajatxo.coral.data.prefs.PlaybackHistory.history.collectAsState()
   885→
   886→    val speedDialSongs = remember(songs.size, pinnedIds, launchSeed, speedDialMode, playbackHistory) {
   887→        if (songs.isEmpty()) emptyList()
   888→        else {
   889→            // 1. Pinned songs (in pin order), filtered to ones that still
   890→            //    exist in the library (in case a pinned song was deleted).
   891→            val pinnedSongs = pinnedIds.mapNotNull { id ->
   892→                songs.firstOrNull { it.id == id }
   893→            }
   894→            // 2. Pool of available (non-pinned) songs.
   895→            val pool = songs.filter { it.id !in pinnedIds }
   896→
   897→            // 3. Pick the songs based on the current mode.
   898→            val targetCount = 26
   899→            val randomCount = (targetCount - pinnedSongs.size).coerceAtLeast(0)
   900→
   901→            val picked: List<Song> = if (speedDialMode == com.rajatxo.coral.data.prefs.SpeedDialModeManager.SpeedDialMode.LAST_PLAYED
   902→                && playbackHistory.isNotEmpty()) {
   903→                // ★ LAST_PLAYED mode — 80% from last-played artist + 20% other.
   904→                //   If the artist has fewer songs than the 80% target, the
   905→                //   remaining slots are filled with random songs from other
   906→                //   artists so all 3 pages (26 tiles) always show.
   907→                val lastArtist = playbackHistory.firstOrNull()?.artist
   908→                if (!lastArtist.isNullOrBlank()) {
   909→                    val artistSongs = pool.filter {
   910→                        it.artist.trim().equals(lastArtist.trim(), ignoreCase = true)
   911→                    }.shuffled(Random(launchSeed + 100))
   912→                    val otherSongs = pool.filter {
   913→                        !it.artist.trim().equals(lastArtist.trim(), ignoreCase = true)
   914→                    }.shuffled(Random(launchSeed + 200))
   915→                    val eightyPct = (randomCount * 0.8f).toInt()
   916→                    // Take up to eightyPct from the artist — if the artist
   917→                    // has fewer songs, take all of them.
   918→                    val artistPicked = artistSongs.take(eightyPct)
   919→                    // Fill the rest of the 80% slot + the 20% slot with
   920→                    // other songs so we always reach randomCount total.
   921→                    val remainingNeeded = randomCount - artistPicked.size
   922→                    artistPicked + otherSongs.take(remainingNeeded)
   923→                } else {
   924→                    pool.shuffled(Random(launchSeed + 100)).take(randomCount)
   925→                }
   926→            } else {
   927→                // RANDOM mode — completely random (original behavior)
   928→                pool.shuffled(Random(launchSeed + 100)).take(randomCount)
   929→            }
   930→            // ★ Failsafe — if even the pool is too small (e.g., tiny library),
   931→            //   pad with whatever songs are available (with possible repeats
   932→            //   from the pool) so all 26 tiles always fill. Without this,
   933→            //   pages would be empty for small libraries.
   934→            val finalPicked = if (picked.size < randomCount && pool.isNotEmpty()) {
   935→                val stillNeeded = randomCount - picked.size
   936→                // Cycle through the pool to fill — repeats are acceptable
   937→                // because the user explicitly wants all pages filled.
   938→                val fillers = pool.shuffled(Random(launchSeed + 300))
   939→                    .filter { it.id !in picked.map(Song::id) }  // avoid dups
   940→                    .take(stillNeeded)
   941→                picked + fillers
   942→            } else {
   943→                picked
   944→            }
   945→            pinnedSongs + finalPicked
   946→        }
   947→    }
   948→
   949→    if (speedDialSongs.isEmpty()) return
   950→
   951→    // Section header — "Speed dial" text + chevron right beside it.
   952→    // ★ Both the text AND the chevron are clickable. Tapping either opens
   953→    //   the floating SpeedDialCard (rendered outside the layerBackdrop
   954→    //   by the caller — same pattern as TodaysTopCard). The card lets the
   955→    //   user swipe to switch Speed Dial mode (Random / Last Played) with
   956→    //   haptics + sound — same UX as the nav bar's tab switcher.
   957→    Box(
   958→        modifier = Modifier
   959→            .fillMaxWidth()
   960→            .padding(start = 4.dp)  // align with first card's content
   961→    ) {
   962→        Row(
   963→            modifier = Modifier.fillMaxWidth(),
   964→            horizontalArrangement = Arrangement.spacedBy(6.dp),
   965→            verticalAlignment = Alignment.CenterVertically
   966→        ) {
   967→            // ★ "Speed dial" text — tap → open SpeedDialCard
   968→            Text(
   969→                text = "Speed dial",
   970→                color = textPrimary,
   971→                fontSize = 20.sp,
   972→                fontWeight = FontWeight.SemiBold,
   973→                fontFamily = CalSansFamily,
   974→                modifier = Modifier.clickable(
   975→                    interactionSource = remember { MutableInteractionSource() },
   976→                    indication = null,
   977→                    onClick = onSpeedDialHeaderClick
   978→                )
   979→            )
   980→            // ★ Chevron right beside the text — also tap → open SpeedDialCard
   981→            Box(
   982→                modifier = Modifier
   983→                    .size(24.dp)
   984→                    .clickable(
   985→                        interactionSource = remember { MutableInteractionSource() },
   986→                        indication = null,
   987→                        onClick = onSpeedDialHeaderClick
   988→                    ),
   989→                contentAlignment = Alignment.Center
   990→            ) {
   991→                Icon(
   992→                    imageVector = CoralIcons.ChevronRight,
   993→                    contentDescription = "Open Speed Dial mode card",
   994→                    tint = textSecondary,
   995→                    modifier = Modifier.size(20.dp)
   996→                )
   997→            }
   998→            // ─── Bug-on-a-line pull-to-refresh indicator ──────────────
   999→            BugLineRefreshIndicator(
  1000→                progress = pullProgress,
  1001→                isRefreshing = isRefreshing,
  1002→                modifier = Modifier
  1003→                    .weight(1f)
  1004→                    .height(20.dp)
  1005→            )
  1006→        }
  1007→    }
  1008→
  1009→    Spacer(modifier = Modifier.height(6.dp))
  1010→
  1011→    // Grid layout: 3 columns, paginated
  1012→    val targetItemSize = 110.dp
  1013→    val columns = 3
  1014→    val rows = 3
  1015→    val itemsPerPage = columns * rows // 9 per page
  1016→    val totalSlots = speedDialSongs.size + 1 // +1 for the dice
  1017→    val pageCount = (totalSlots + itemsPerPage - 1) / itemsPerPage
  1018→    val pagerState = rememberPagerState(pageCount = { pageCount.coerceAtLeast(1) })
  1019→
  1020→    val itemWidth = targetItemSize
  1021→
  1022→    Column(
  1023→        modifier = Modifier.fillMaxWidth()
  1024→    ) {
  1025→        // ★ Glass morphism box wrapping ONLY the 3x3 grid (not header, not page indicator).
  1026→        // Uses Haze (dev.chrisbanes.haze) — same library as ArchiveTune/BitChord.
  1027→        // When glassHazeState is provided (Cynthia), uses hazeEffect for real glass.
  1028→        // When null (Astra), renders normally without glass.
  1029→        val gridShape = RoundedCornerShape(20.dp)
  1030→        val gridModifier = if (glassHazeState != null && glassStyle != null) {
  1031→            Modifier
  1032→                .fillMaxWidth()
  1033→                .height(itemWidth * rows + 16.dp)
  1034→                .clip(gridShape)
  1035→                .hazeEffect(
  1036→                    state = glassHazeState,
  1037→                    style = glassStyle
  1038→                )
  1039→                .border(1.dp, Color.White.copy(alpha = 0.1f), gridShape)
  1040→                .padding(8.dp)
  1041→        } else {
  1042→            Modifier
  1043→                .fillMaxWidth()
  1044→                .height(itemWidth * rows + 16.dp)
  1045→        }
  1046→
  1047→        HorizontalPager(
  1048→            state = pagerState,
  1049→            contentPadding = PaddingValues(horizontal = 0.dp),
  1050→            pageSpacing = 12.dp,
  1051→            beyondViewportPageCount = 1,
  1052→            modifier = gridModifier
  1053→        ) { page ->
  1054→            Column(modifier = Modifier.fillMaxSize()) {
  1055→                for (row in 0 until rows) {
  1056→                    Row(modifier = Modifier.fillMaxWidth()) {
  1057→                        for (col in 0 until columns) {
  1058→                            val itemIndex = row * columns + col
  1059→                            val globalItemIndex = page * itemsPerPage + itemIndex
  1060→
  1061→                            // The dice button is the last slot on the first page
  1062→                            val isDiceSlot = (globalItemIndex == itemsPerPage - 1)
  1063→
  1064→                            if (isDiceSlot) {
  1065→                                RandomizeGridItem(
  1066→                                    isLoading = isRandomizing,
  1067→                                    onClick = {
  1068→                                        if (isRandomizing) {
  1069→                                            isRandomizing = false
  1070→                                        } else {
  1071→                                            isRandomizing = true
  1072→                                            scope.launch {
  1073→                                                kotlinx.coroutines.delay(800)  // dice animation
  1074→                                                val randomSong = songs.random()
  1075→                                                isRandomizing = false
  1076→                                                onSongClick(randomSong)
  1077→                                            }
  1078→                                        }
  1079→                                    },
  1080→                                    modifier = Modifier
  1081→                                        .width(itemWidth)
  1082→                                        .height(itemWidth)
  1083→                                        .padding(4.dp)
  1084→                                )
  1085→                            } else {
  1086→                                val actualIndex = if (globalItemIndex < itemsPerPage - 1) {
  1087→                                    globalItemIndex
  1088→                                } else {
  1089→                                    globalItemIndex - 1
  1090→                                }
  1091→                                val song = speedDialSongs.getOrNull(actualIndex)
  1092→                                if (song != null) {

  1500→    // ═══ Particle burst randomize button ═══
  1501→    // Idle: a soft glowing dot in the center, gently pulsing.
  1502→    // On tap: 14 particles burst outward in all directions, each with a
  1503→    // gradient color (coral red → orange → pink → gold), shrinking +

  1507→    // SpeedDialSection sets isRandomizing=true on tap, waits 800ms,
  1508→    // picks a random song, sets isRandomizing=false). The burst lasts
  1509→    // 600ms — the remaining 200ms is a "settle" moment where the idle
  1510→    // dot reappears before the song plays.
  1511→
  1512→    // ─── Particle data (precomputed once, stable across recompositions) ──

  1516→            Color(0xFFFF6B6B),  // coral red
  1517→            Color(0xFFFF9F40),  // orange
  1518→            Color(0xFFFF4081),  // pink
  1519→            Color(0xFFFFD700)   // gold
  1520→        )

  1522→            Particle(
  1523→                angle = (i * 360.0 / 14.0 + kotlin.random.Random.nextFloat() * 20.0) * (kotlin.math.PI / 180.0),
  1524→                color = colors[i % colors.size],
  1525→                sizeMult = 0.7f + kotlin.random.Random.nextFloat() * 0.6f,
  1526→                speedMult = 0.8f + kotlin.random.Random.nextFloat() * 0.4f
  1527→            )
  1528→        }

  1533→    val pulseScale by infiniteTransition.animateFloat(
  1534→        initialValue = 0.9f,
  1535→        targetValue = 1.1f,
  1536→        animationSpec = infiniteRepeatable(
  1537→            animation = tween(1500, easing = androidx.compose.animation.core.FastOutSlowInEasing),
  1538→            repeatMode = RepeatMode.Reverse
  1539→        ),
  1540→        label = "pulseScale"
  1541→    )
  1542→
  1543→    // ─── Burst animation (triggered by isLoading) ──
  1544→    val burstProgress = remember { androidx.compose.animation.core.Animatable(0f) }
  1545→    androidx.compose.runtime.LaunchedEffect(isLoading) {

  1551→            )

  1557→            .clip(RoundedCornerShape(8.dp))
  1558→            .background(Color.White.copy(alpha = 0.08f))

  1562→                onClick = onClick
  1563→            ),

  1565→    ) {
  1566→        Canvas(modifier = Modifier.fillMaxSize()) {
  1567→            val center = androidx.compose.ui.geometry.Offset(size.width / 2f, size.height / 2f)
  1568→            val dotRadius = size.minDimension * 0.12f
  1569→
  1570→            // ─── Idle glowing dot (always visible, pulsing) ──
  1571→            // 3 concentric circles for a soft glow effect
  1572→            val glowRadius = dotRadius * pulseScale
  1573→            drawCircle(
  1574→                color = Color(0xFFFF6B6B).copy(alpha = 0.08f),
  1575→                radius = glowRadius * 2.8f,
  1576→                center = center
  1577→            )
  1578→            drawCircle(
  1579→                color = Color(0xFFFF6B6B).copy(alpha = 0.15f),
  1580→                radius = glowRadius * 2.0f,
  1581→                center = center
  1582→            )
  1583→            drawCircle(
  1584→                color = Color(0xFFFF6B6B),
  1585→                radius = glowRadius,
  1586→                center = center
  1587→            )
  1588→
  1589→            // ─── Burst particles (visible during burst animation) ──
  1590→            val progress = burstProgress.value
  1591→            if (progress > 0f && progress < 1f) {
  1592→                val maxDistance = size.minDimension * 0.45f
  1593→                particles.forEach { p ->
  1594→                    val distance = maxDistance * progress * p.speedMult
  1595→                    val x = (center.x + kotlin.math.cos(p.angle) * distance).toFloat()
  1596→                    val y = (center.y + kotlin.math.sin(p.angle) * distance).toFloat()
  1597→                    val particleSize = dotRadius * p.sizeMult * (1f - progress)
  1598→                    val alpha = (1f - progress) * 0.9f
  1599→
  1600→                    drawCircle(
  1601→                        color = p.color.copy(alpha = alpha),
  1602→                        radius = particleSize,
  1603→                        center = androidx.compose.ui.geometry.Offset(x, y)
